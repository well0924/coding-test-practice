- 만약에 정수 타입을 받아서 자릿수를 쪼갠다면 해당 정수를 10진수로 나눠서 다시 배열에 담아야 한다는 번거로움이 있기에 문자열 타입을 선택을 하게 되었습니다.

public static void main(Stirng[]args) throw IOException {
	BuffererReader br = new BuffererdReader(new InputStreamReader(System.in));
	// 입력값 
	String input = br.readLine();
	
	// 입력값을 Char타입의 배열로 전환을 하기.
	char[] arr = input.ChartoArray();
	// char배열의 길이만큼의 결괏값을 출력할 배열을 생성
	char[] result = new char[arr.length];
	// 입력값의 자릿수를 오름차순으로 변경
	Arrays.sort(arr);
	
	for(int i = 0; i< arr.lenght(); i++) {
		// arr의 배열에서 역순으로 result배열에 넣기.
		result[i] = arr[arr.length-1-i];
	}
	// 결과값을 출력하기
	System.out.print(result);
}
